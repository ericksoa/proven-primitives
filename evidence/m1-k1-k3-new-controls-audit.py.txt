from pathlib import Path
import datetime,hashlib,json,re
rows=json.loads(Path('evidence/m1-k1-k3-controls-results.json').read_text());freeze=json.loads(Path('evidence/m1-k1-k3-attempt02-source-freeze.json').read_text());assert freeze['exit_code']==0 and freeze['source_unchanged']
checked=[]
for row in rows:
 assert row['exit_code']==0 and row['source_unchanged'] and datetime.datetime.fromisoformat(row['started_utc'])>datetime.datetime.fromisoformat(freeze['finished_utc'])
 names=re.findall(r'^REJECTED (brotli-[a-z0-9-]+):',Path(row['log']).read_text(),re.MULTILINE);assert len(names)==row['expected_mutants']
 for name in names:
  path=Path('evidence')/('canary-'+name+'.log');text=path.read_text();low=text.lower()
  assert re.search(r'verification results:: .*\b[1-9][0-9]* errors',text),name
  assert any(x in low for x in ['postcondition not satisfied','precondition not satisfied','invariant not satisfied','assertion failed']),name
  assert not any(x in low for x in ['resource limit','rlimit) exceeded','expected rlimit-count','panicked','internal compiler error','unexpected smt output']),name
  checked.append({'name':name,'log':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'semantic_rejection':True})
assert len(checked)==8
Path('evidence/m1-k1-k3-new-controls-audit.json').write_text(json.dumps({'mutants':checked,'whole_positive':'evidence/brotli-b3c-m1-k1-k3-attempt02-verification.json','resource_or_tool_failures_counted':0},indent=2)+'\n')
print('Eight retained transcripts independently confirm semantic rejection, with no resource or tool failure.')
