from pathlib import Path
import concurrent.futures,datetime,hashlib,json,os,re,subprocess,time
freeze=json.loads(Path('evidence/m1-full-codec-attempt02-source-freeze.json').read_text())
proof=json.loads(Path('evidence/brotli-b3c-m1-full-codec-attempt02-verification.json').read_text())

identities=dict(freeze['source_sha256'],**freeze['validation_source_sha256'])
def unchanged():return all(hashlib.sha256(Path(n).read_bytes()).hexdigest()==h for n,h in identities.items())
assert unchanged()
env=dict(os.environ,DEVELOPER_DIR='/Library/Developer/CommandLineTools',RUSTUP_TOOLCHAIN='1.99.1',VERUS_ROOT='/Users/aerickson/Documents/NemoDev/formal-verification-pilot-2026-10-23/tools/verus-arm64-macos')
source=Path('tests/proof_rejection.rs').read_text()
cases=[('the_prover_rejects_four_m1_tree_action_bound_and_drain_defects',4)]
def run(case):
 name,expected=case;start=source.index('fn '+name+'(');end=start+source[start:].index('\n}\n')+3;names=list(dict.fromkeys(re.findall(r'"(brotli-[a-z0-9-]+)"',source[start:end])));assert len(names)==expected
 log=Path('evidence')/('m1-full-codec-attempt02-controls-'+name+'.log');clock=time.monotonic();started=datetime.datetime.now(datetime.timezone.utc).isoformat()

 with log.open('w') as out:result=subprocess.run(cmd,env=env,stdout=out,stderr=subprocess.STDOUT)
 row={'suite':name,'expected_mutants':expected,'exit_code':result.returncode,'started_utc':started,'seconds':time.monotonic()-clock,'log':str(log),'sha256':hashlib.sha256(log.read_bytes()).hexdigest(),'source_unchanged':unchanged(),'mutants':[]}
 if result.returncode==0:
  for mutant in names:
   p=Path('evidence')/('canary-'+mutant+'.log');text=p.read_text();lower=text.lower()

   row['mutants'].append({'name':mutant,'log':str(p),'sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'semantic_rejection':True})
 print(json.dumps(row),flush=True);return row
