#[verifier::spinoff_prover]
#[verifier::rlimit(1000)]
fn r_triage_context_cursor_owner_call(
    workspace: &mut [u8],
    window_bytes: u32,
    budget: &mut crate::brotli::RequestBudget,
    input: &[u8],
    end_of_input: bool,
) -> (r: (Result<usize, CompressedStep>, Ghost<nat>))
    requires
        old(budget).valid(),
        b3c_workspace_shape(old(workspace)@.len() as nat, window_bytes),
        compressed_decoder_state_ok(old(workspace)@, window_bytes),
        window_bytes == 1_024,
        old(budget).work_remaining == 5,
        input.len() == 0, !end_of_input,
        old(workspace)@.len() == 1_177_584,
        forall|i: int| 0 <= i < old(workspace)@.len() ==> #[trigger] old(workspace)@[i] ==
            if i == 1_164_308 { 9u8 }
            else if i == 1_164_310 { 5u8 }
            else if i == 1_164_316 { 1u8 }
            else if i == 1_164_459 { 63u8 } else { 0u8 },
    ensures
        r.1@ == 5,
        final(budget).work_remaining == 0,
        serialized_terminal_error(final(workspace)@, window_bytes, CompressedHeaderError::InternalState),
        terminal_error_frame(old(workspace)@, final(workspace)@, window_bytes),
        r.1@ + final(budget).work_remaining as nat == old(budget).work_remaining as nat,

        m1_context_bad_saved_cursor(old(workspace)@, window_bytes) ==>
            m1_context_cursor_owner_effect(old(workspace)@, final(workspace)@, window_bytes,
                *old(budget), *final(budget), r.0),
        b3b_scalar_byte_at(old(workspace)@, window_bytes, B3B_OUTER_PHASE) == 3
            && m1_bad_initial_suffix(old(workspace)@, window_bytes) ==>
                m1_bad_initial_suffix_owner_effect(old(workspace)@, final(workspace)@, window_bytes,
                    *old(budget), *final(budget), r.0),
        b3b_scalar_byte_at(old(workspace)@, window_bytes, B3B_OUTER_PHASE) == 3
            && initial_suffix_read_domain(old(workspace)@, window_bytes) ==>
            exists|plan: CompressedActionPlan| #[trigger] initial_suffix_owner_effect(
                old(workspace)@, final(workspace)@, window_bytes,
                *old(budget), *final(budget), input@, end_of_input, r.0, plan),
        fixed_outer_bad_cursor(old(workspace)@, window_bytes) ==> {
            &&& final(budget).input_remaining == old(budget).input_remaining
            &&& final(budget).output_remaining == old(budget).output_remaining
            &&& final(budget).calls_remaining == old(budget).calls_remaining
            &&& final(budget).initializations_remaining == old(budget).initializations_remaining
            &&& if old(budget).work_remaining >= 5 {
                &&& r.0 == Err(CompressedStep { consumed: 0,
                    status: CompressedStatus::Failed(CoreFailure::InvalidFormat(CoreFormatDetail::StateEncoding)) })
                &&& final(budget).work_remaining == old(budget).work_remaining - 5
                &&& terminal_error_frame(old(workspace)@, final(workspace)@, window_bytes)
                &&& serialized_terminal_error(final(workspace)@, window_bytes, CompressedHeaderError::InternalState)
            } else {
                &&& r.0 == Err(CompressedStep { consumed: 0, status: CompressedStatus::Failed(CoreFailure::ResourceLimit) })
                &&& final(workspace)@ == old(workspace)@
                &&& *final(budget) == *old(budget)
            }
        },
        b3b_scalar_byte_at(old(workspace)@, window_bytes, B3B_OUTER_PHASE) == 5
            && context_mode_handoff_domain(old(workspace)@, window_bytes) ==> {
                &&& final(budget).input_remaining == old(budget).input_remaining
                &&& if old(budget).work_remaining >= 18 {
                    &&& r.0 == Ok(0usize)
                    &&& final(budget).work_remaining == old(budget).work_remaining - 18
                    &&& context_mode_handoff_applied(old(workspace)@, final(workspace)@, window_bytes)
                } else {
                    &&& r.0 == Err(CompressedStep { consumed: 0, status: CompressedStatus::Failed(CoreFailure::ResourceLimit) })
                    &&& *final(budget) == *old(budget)
                    &&& final(workspace)@ == old(workspace)@
                }
            },
        b3b_scalar_byte_at(old(workspace)@, window_bytes, B3B_OUTER_PHASE) == 5
            && context_mode_count_invalid(old(workspace)@, window_bytes) ==> {
                &&& final(budget).input_remaining == old(budget).input_remaining
                &&& final(budget).output_remaining == old(budget).output_remaining
                &&& final(budget).calls_remaining == old(budget).calls_remaining
                &&& final(budget).initializations_remaining == old(budget).initializations_remaining
                &&& if old(budget).work_remaining >= 5 {
                    &&& r.0 == Err(CompressedStep { consumed: 0,
                        status: CompressedStatus::Failed(CoreFailure::InvalidFormat(CoreFormatDetail::StateEncoding)) })
                    &&& final(budget).work_remaining == old(budget).work_remaining - 5
                    &&& terminal_error_frame(old(workspace)@, final(workspace)@, window_bytes)
                    &&& serialized_terminal_error(final(workspace)@, window_bytes, CompressedHeaderError::InternalState)
                } else {
                    &&& r.0 == Err(CompressedStep { consumed: 0, status: CompressedStatus::Failed(CoreFailure::ResourceLimit) })
                    &&& *final(budget) == *old(budget)
                    &&& final(workspace)@ == old(workspace)@
                }
            },
        final(budget).valid(),
        final(budget).limits == old(budget).limits,
        final(budget).work_remaining <= old(budget).work_remaining,
        final(workspace)@.len() == old(workspace)@.len(),
        match r.0 {
            Ok(consumed) => {
                &&& consumed <= input.len()
                &&& compressed_decoder_state_ok(final(workspace)@, window_bytes)
            },
            Err(step) => {
                &&& step.consumed <= input.len()
                &&& (!matches!(step.status, CompressedStatus::Failed(_)) ==>
                    compressed_decoder_state_ok(final(workspace)@, window_bytes))

            &input[..initial_allowed],
            end_of_input && initial_allowed == input.len(),
        ); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {
            Ok(value) => value,
            Err(_) => {
                return (Err(CompressedStep {
                    consumed: 0,
                    status: CompressedStatus::Failed(core_state_failure()),
                }), Ghost(r_header_action_source_units));
            }
        };
        proof {
            if context_mode_pending(workspace@, window_bytes) {
                reveal(context_mode_plan_valid);
                context_mode_planning_reads_agree(workspace@, window_bytes, initial_allowed as nat,
                    planning_work, plan.planning_work);
            }
        }
        if plan.planning_work != planning_work {
            return (Err(CompressedStep { consumed: 0,
                status: CompressedStatus::Failed(core_state_failure()) }), Ghost(r_header_action_source_units));
        }
        match { let (r_child_result, Ghost(r_child_units)) = execute_fixed_compressed_plan(
            workspace,
            window_bytes,
            budget,
            CompressedOuterPhase::ContextModes,
            &input[..initial_allowed],
            end_of_input && initial_allowed == input.len(),
            initial_allowed < input.len(),
            plan,
        ); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {
            Ok(used) => already_consumed = used,
            Err(step) => return (Err(step), Ghost(r_header_action_source_units)),
        }
    } else if (initial_outer == 6 || initial_outer == 7) && initial_context_phase == 0 {
        proof {
            assert(6 <= b3b_scalar_byte_at(
                workspace@, window_bytes, B3B_OUTER_PHASE,
            ) <= 7);
        }
        match { let (r_child_result, Ghost(r_child_units)) = advance_planned_context_tree_count(
            workspace, window_bytes, budget, input, end_of_input,
        ); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {
            Ok(used) => already_consumed = used,
            Err(step) => return (Err(step), Ghost(r_header_action_source_units)),
        }
    } else if (initial_outer == 6 || initial_outer == 7)
        && (initial_context_phase == 1 || initial_context_phase == 2
            || initial_context_phase == 5 || initial_context_phase == 6)
    {
        match { let (r_child_result, Ghost(r_child_units)) = advance_planned_context_bits(workspace, window_bytes, budget, input, end_of_input); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {
            Ok(used) => already_consumed = used,
            Err(step) => return (Err(step), Ghost(r_header_action_source_units)),
        }
    } else if (initial_outer == 6 || initial_outer == 7) && initial_context_phase == 3 {
        match { let (r_child_result, Ghost(r_child_units)) = advance_planned_context_tree(workspace, window_bytes, budget, input, end_of_input); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {
            Ok(used) => already_consumed = used,
            Err(step) => return (Err(step), Ghost(r_header_action_source_units)),
        }
    } else if (initial_outer == 6 || initial_outer == 7) && initial_context_phase == 4 {
        match { let (r_child_result, Ghost(r_child_units)) = advance_planned_context_symbol(workspace, window_bytes, budget, input, end_of_input); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {
            Ok(used) => already_consumed = used,
            Err(step) => return (Err(step), Ghost(r_header_action_source_units)),
        }
    } else if (initial_outer == 6 || initial_outer == 7)
        && 7 <= initial_context_phase && initial_context_phase <= 9
    {
        let phase = if initial_outer == 6 {
            CompressedOuterPhase::LiteralContextMap
        } else {
            CompressedOuterPhase::DistanceContextMap
        };
        proof {
            reveal(compressed_outer_phase_tag_spec);
            assert(b3b_scalar_byte_at(
                workspace@,
                window_bytes,
                B3B_OUTER_PHASE,
            ) == compressed_outer_phase_tag_spec(phase));
            assert(7 <= b3b_scalar_byte_at(
                workspace@,
                window_bytes,
                B3B_CONTEXT_PHASE,
            ) <= 9);
        }
        match { let (r_child_result, Ghost(r_child_units)) = advance_planned_context_map_tail(
            workspace,
            window_bytes,
            budget,
            phase,
            input,
            end_of_input,
        ); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {
            Ok(used) => already_consumed = used,
            Err(step) => return (Err(step), Ghost(r_header_action_source_units)),
        }
    } else if initial_outer == 8 || initial_outer == 9 || initial_outer == 10 {
        let phase = if initial_outer == 8 {
            CompressedOuterPhase::LiteralTreeGroup
        } else if initial_outer == 9 {
            CompressedOuterPhase::CommandTreeGroup
        } else {
            CompressedOuterPhase::DistanceTreeGroup
        };
        proof {
            reveal(compressed_outer_phase_tag_spec);
            assert(b3b_scalar_byte_at(
                workspace@,
                window_bytes,
                B3B_OUTER_PHASE,
            ) == compressed_outer_phase_tag_spec(phase));
        }
        match { let (r_child_result, Ghost(r_child_units)) = advance_planned_tree_group(
            workspace,
            window_bytes,
            budget,
            phase,
            input,
            end_of_input,
        ); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {
            Ok(used) => already_consumed = used,
            Err(step) => return (Err(step), Ghost(r_header_action_source_units)),
        }
    } else if initial_outer < 11 {
        // W1 rejects unknown initial-length/context subphases without reading
        // input. Preserve that terminal action and prepay its four writes.
        let ghost fallback_before = workspace@;
        let ghost fallback_work = budget.work_remaining;
        proof { m1_suffix_fallback_image_is_admitted(); }
        if !budget.charge_work(4 + 1) {
            proof {
                if initial_outer == 3 && m1_bad_initial_suffix(old(workspace)@, window_bytes) {
                    reveal(m1_bad_initial_suffix_validation_reads); reveal(m1_bad_initial_suffix_owner_effect);
                    assert(m1_bad_initial_suffix_validation_reads(old(workspace)@, window_bytes) == 0);
                    assert(m1_bad_initial_suffix_owner_effect(old(workspace)@, workspace@, window_bytes,
                        *old(budget), *budget,
                        Err(CompressedStep { consumed: 0, status: CompressedStatus::Failed(CoreFailure::ResourceLimit) })));
                }
            }
            return (Err(CompressedStep { consumed: 0,
                status: CompressedStatus::Failed(CoreFailure::ResourceLimit) }), Ghost(r_header_action_source_units));
        }
        proof {
            if window_bytes == 1024 && fallback_before == m1_suffix_fallback_image() && fallback_work == 5 {
                assert(budget.work_remaining == 0);
            }
        }
        proof { r_header_action_source_units = r_header_action_source_units + 1; }
        if { let (r_child_result, Ghost(r_child_units)) = store_terminal_error(workspace, window_bytes, CompressedHeaderError::InternalState); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result }.is_err() {
            return (Err(CompressedStep { consumed: 0,
                status: CompressedStatus::Failed(core_state_failure()) }), Ghost(r_header_action_source_units));
        }
        proof {
            if window_bytes == 1024 && fallback_before == m1_suffix_fallback_image() && fallback_work == 5 {
                assert(workspace@[b3b_scalar_base(1024) + 175] == 0);
            }
        }
        proof {
            if initial_outer == 3 && m1_bad_initial_suffix(old(workspace)@, window_bytes) {
                reveal(m1_bad_initial_suffix_validation_reads); reveal(m1_bad_initial_suffix_owner_effect);
                assert(m1_bad_initial_suffix_validation_reads(old(workspace)@, window_bytes) == 0);
                assert(m1_bad_initial_suffix_owner_effect(old(workspace)@, workspace@, window_bytes,
                    *old(budget), *budget,
                    Err(CompressedStep { consumed: 0,
                        status: CompressedStatus::Failed(CoreFailure::InvalidFormat(CoreFormatDetail::StateEncoding)) })));
            }
        }
        return (Err(CompressedStep { consumed: 0,
            status: CompressedStatus::Failed(core_state_failure()) }), Ghost(r_header_action_source_units));
    } else {
        proof { assert(initial_outer >= 11); }
        return (Err({ let (r_child_result, Ghost(r_child_units)) = advance_planned_terminal_header(workspace, window_bytes, budget, initial_outer); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result }), Ghost(r_header_action_source_units));
    }
    let ordinary_result = {
        Ok(already_consumed)
    };
    (ordinary_result, Ghost(r_header_action_source_units))
}
