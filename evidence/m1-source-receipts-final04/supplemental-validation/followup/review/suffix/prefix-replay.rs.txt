#[verifier::spinoff_prover]
#[verifier::rlimit(1000)]
fn r_triage_suffix_fallback_prefix_replay(
    workspace: &mut [u8], window_bytes: u32,
    budget: &mut crate::brotli::RequestBudget, input: &[u8], end_of_input: bool,
) -> (r: (bool, Ghost<nat>))
    requires
        old(budget).valid(), old(budget).work_remaining == 5,
        window_bytes == 1024, old(workspace)@ == m1_suffix_fallback_image(),
        input.len() == 0, !end_of_input,
    ensures
        r.0,
        r.1@ == 0,
        final(workspace)@ == old(workspace)@,
        *final(budget) == *old(budget),
        final(budget).valid(),
        final(budget).work_remaining == 5,

    hide(serialized_mtf_state_ok);
    hide(serialized_context_state_ok);

    proof {

        }
    } else if (initial_outer == 6 || initial_outer == 7) && initial_context_phase == 4 {
        match { let (r_child_result, Ghost(r_child_units)) = advance_planned_context_symbol(workspace, window_bytes, budget, input, end_of_input); proof { r_header_action_source_units = r_header_action_source_units + r_child_units; } r_child_result } {

        }
    } else if (initial_outer == 6 || initial_outer == 7)
        && 7 <= initial_context_phase && initial_context_phase <= 9
    {
        let phase = if initial_outer == 6 {
            CompressedOuterPhase::LiteralContextMap
        } else {
            CompressedOuterPhase::DistanceContextMap
        };
        proof {
            reveal(compressed_outer_phase_tag_spec);
            assert(b3b_scalar_byte_at(

        } else if initial_outer == 9 {

        } else {

        }
    } else if initial_outer < 11 {
        // W1 rejects unknown initial-length/context subphases without reading
        // input. Preserve that terminal action and prepay its four writes.
        let ghost fallback_before = workspace@;
        let ghost fallback_work = budget.work_remaining;
        proof { m1_suffix_fallback_image_is_admitted(); }
